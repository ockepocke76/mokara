# Note: numpy (as np) is assumed to be available in the execution environment.

class T3(BaseStrategy):

    @property
    def parameters(self):
        return {
            "accumulation_contribution": {
                "label": "Annual Contribution (Real)",
                "default": 120000.0,
                "description": "Annual investment contribution during the accumulation phase (in Year 0 real terms)."
            },
            "target_net_worth": {
                "label": "Target Net Worth (Real)",
                "default": 10000000.0,
                "description": "Net worth target (in Year 0 real terms) that triggers the switch from accumulation to retirement phase."
            },
            "withdrawal_rate": {
                "label": "Initial Withdrawal Rate",
                "default": 0.04,
                "description": "Percentage of the inflation-adjusted transition net worth to be withdrawn annually in retirement phase."
            }

    @property
    def shortfall_funding_policy(self) -> list[str]:
        # Standard policy for withdrawal strategies: fund necessary cash (drawdown/costs) by selling assets,
        # using existing cash only to cover shortfalls caused by tax drag, if necessary.
        return ['SELL_ASSETS', 'USE_CASH']

    def initialize_portfolio(self, initial_portfolio_state: dict, market_data_at_start: pd.DataFrame) -> dict:
        # Invest all initial cash immediately.
        cash = initial_portfolio_state.get('cash', 0.0)
        return {'action': 'BUY_ASSET', 'cash_amount': cash}

    def _determine_phase_and_transition_value(self, portfolio_history: list[dict]):
        """Determines if the transition target has been met and returns the nominal value and year if so."""
        
        target_real = self.params.get('target_net_worth', 10000000.0)
        R = self.params.get('inflation_rate', 0.03)

        transition_occurred = False
        nominal_transition_value = 0.0
        transition_year = 0
        
        # Iterate through history (Year 1 onwards, since Year 0 is just initialization)
        # Assuming np is available for power calculations.
        for entry in portfolio_history:
            year = entry['year']
            if year == 0:
                continue
            
            current_nominal_net_worth = entry['net_worth']
            
            # Calculate the nominal target net worth for this historical year (Y)
            # Inflation factor converts Year 0 real value to Year Y nominal value.
            inflation_factor = np.power(1.0 + R, year)
            target_nominal = target_real * inflation_factor
            
            if current_nominal_net_worth >= target_nominal:
                transition_occurred = True
                nominal_transition_value = current_nominal_net_worth 
                transition_year = year
                
                # Stop searching once transition is found (we only care about the *first* trigger)
                break

        return transition_occurred, nominal_transition_value, transition_year


    def get_annual_drawdown(self, year: int, portfolio_state: dict, portfolio_history: list[dict]) -> float:
        
        transition_occurred, V_T, T = self._determine_phase_and_transition_value(portfolio_history)
        
        if not transition_occurred:
            # Accumulation Phase: No drawdown for consumption
            return 0.0
        
        # Retirement Phase
        withdrawal_rate = self.params.get('withdrawal_rate', 0.04)
        R = self.params.get('inflation_rate', 0.03)

        # Calculate inflation adjustment since transition (T) up to the current year (Y).
        # Drawdown for Year Y (current `year` parameter) is based on inflation adjustment Y - T.
        inflation_exponent = year - T
        
        # Safety check: Drawdown should only happen after transition year T.
        if inflation_exponent <= 0:
            return 0.0 

        # Calculate inflation factor for maintaining real value
        inflation_factor = np.power(1.0 + R, inflation_exponent)
        
        # Nominal withdrawal amount: V_T * Rate * Inflation Adjustment
        drawdown = V_T * withdrawal_rate * inflation_factor
        
        return drawdown

            total_cash_needed = desired_drawdown + mandatory_costs

            amount_sold = total_cash_needed

            'amount_sold': amount_sold,
